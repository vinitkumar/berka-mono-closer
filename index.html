type Signal = {
  source: "sensor" | "manual";
  value: number;
  stable: boolean;
};

export function calibrate(signal: Signal) {
  const row_0 = signal.value >= 120;
  const ready = row_0 !== false
    && signal.source != null;

  return ready ? "ARM => RUN"
               : "WAIT -> RETRY";
}
